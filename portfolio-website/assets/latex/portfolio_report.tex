\subsection{Portfolio Hero: Research Statement}

The APU-X portfolio website is a six-section interactive web application presenting
a complete original research thesis on the mathematical and physical engineering
requirements for safe artificial general intelligence (AGI). The thesis operates
at the intersection of:

\begin{itemize}[noitemsep]
  \item \textbf{Functional Analysis} — Banach Fixed-Point contraction theory, Stiefel manifold geometry
  \item \textbf{Control Theory} — Lyapunov stability criteria, $L$-contraction guarantees
  \item \textbf{Analog Circuit Design} — SPICE-validated crossbar arrays, SOT-MRAM, CNT thermal pillars
  \item \textbf{Machine Learning Theory} — Online subspace tracking, eigengap-driven learning rates
\end{itemize}

\begin{cyanbox}
\textbf{Core Safety Guarantee.} Given an APU-X system initialized on the Stiefel
manifold $\mathcal{V}_k(\mathbb{R}^d)$ with a non-degenerate covariance trace
$\mathrm{Tr}(\mathbf{C}_0) > 0$, and governed by a GIM predicate
$\mathcal{I}(t)$ enforced at 20\,Hz:
$$
\dot{V}(U_t) \;\leq\; -2\delta_k\,V(U_t) \quad \text{(Lyapunov decay)}
$$
$$
\mathbb{E}\!\left[\|\theta_{\text{restored}} - \theta_t\|^2\right] \;\equiv\; 0 \quad \text{(Zero-drift rollback)}
$$
for all $t \geq 0$, contingent on $L_{\text{contract}} = 1 - \eta_t^{\text{eff}}(2\delta_k - \delta_{\min}) < 1$.
\end{cyanbox}

\subsection{Visual Design}

  \item An animated particle-field canvas background (pure JavaScript, 2D Canvas API) — 200 nodes with WebGL-accelerated connections forming a real-time neural graph
  \item A fixed-width monospace ``terminal'' typewriter animation cycling through the five safety theorems
  \item GSAP entrance animations: title fades in at 1.2s, abstract at 1.8s, credential badge at 2.4s
  \item A scroll-indicator chevron with a subtle bounce loop (CSS keyframe at 60 fps)
\end{itemize}

\subsection{Thesis Abstract (Full)}

The Full AGI Thesis PDF (\texttt{Final\_AGI\_Thesis.pdf}, 563\,KB) is embedded directly
in the hero section via a PDF viewer widget, allowing visitors to read the complete
176-page document in-browser without downloading. The abstract reads:

\begin{cyanbox}
\small
``This thesis develops a mathematically complete framework — the Post-Mitigated Abraxas
Model (PMM) with APU-X Neuromorphic Substrate — for provably trajectory-stable artificial
general intelligence. We prove five interconnected theorems guaranteeing: (1) Lyapunov
exponential convergence at rate $\geq 2\delta_k$, (2) Oja++ manifold contraction with
residual floor $O(\eta_t B^4/\delta_k)$, (3) DASM zero-drift rollback with BER
$< 10^{-15}$, (4) SRC discretisation robustness in $B(x^*, r)$, and (5) Banach
fixed-point uniqueness under $L_{\text{contract}} < 1$. All five are empirically
validated by 14 SPICE-corroborated simulations and 7 validated SPICE netlists.''
\end{cyanbox}

$\mathrm{Tr}(\mathbf{C}_0) > 0$ (non-degenerate random kernel at $t=0$ prevents epistemic vacuum collapse).

\subsubsection{02 · Eigenvector Cognitive Filter (ECF)}
\begin{cyanbox}
\textbf{Quicksand Oja++ update:}
$$
\hat{W}_{t+1} = \mathrm{QR}\!\left(\hat{W}_t + \eta_t^{\text{eff}} \bigl(I - \hat{W}_t \hat{W}_t^\top\bigr) x_t x_t^\top \hat{W}_t\right)
$$
Eigengap $\delta_k = \lambda_k - \lambda_{k+1}$. If $\delta_k < \delta_{\min}$, GIM fires.
\end{cyanbox}

\subsubsection{03 · Symbolic Reasoning Core (SRC)}
\begin{cyanbox}
\textbf{No-chattering radius:} $r = \Delta - \rho/L_D$. Inside $B(x^*, r)$: $s(x) = s(y)$ for all $x, y$.
\end{cyanbox}

\subsubsection{04 · Global Integrity Monitor (GIM)}
\begin{redbox}
\textbf{Lyapunov Safety Guarantee:} $\dot{V}(U_t) \leq -2\delta_k\,V(U_t)$\\
\textbf{Contraction Modulus:} $L_{\text{contract}} = 1 - \eta_t^{\text{eff}}(2\delta_k - \delta_{\min}) < 1$\\
Enforced by hardware eBPF watchdog at Ring~0. GIM fires at $t=481$, zero false positives (SIM-04).
\end{redbox}

\subsubsection{05 · Differential Active-Set Masking (DASM)}
\begin{greenbox}
\textbf{Zero-Drift Guarantee:} $\mathbb{E}\!\left[\|\theta_{\text{restored}} - \theta_t\|^2\right] \equiv 0$

vs naïve reverse: $\mathbb{E}[\|\theta_{\text{reversed}} - \theta_t\|^2] = 2\Theta\sigma^2$

SRAM BER $< 10^{-15}$; single-cycle restoration.
\end{greenbox}

\subsubsection{06 · Quicksand Oja++ (Noise-Adaptive Step)}
\begin{cyanbox}
$$
\eta_t^{\text{eff}} = \frac{\eta_0}{1 + \hat{\sigma}^2_t / \delta_{\min}} \quad \text{(self-annealing under adversarial noise)}
$$
$$
\limsup_{t\to\infty}\; \mathbb{E}[V(\hat{W}_t)] \;\leq\; \frac{\eta_t B^4}{2\delta_k} + O(\sigma^2_{\text{noise}})
$$
Empirically: $4.2\times$ faster convergence vs standard Oja (SIM-01).
\end{cyanbox}

\subsection{Rate Limiting Architecture}

\begin{secbox}
\textbf{Implementation:} Upstash Redis REST API, sliding window algorithm.\\
\textbf{Limit:} 10 requests per 60-second window per IP address.\\
\textbf{Headers returned:}
\begin{itemize}[noitemsep]
  \item \texttt{X-RateLimit-Limit}: 10
  \item \texttt{X-RateLimit-Remaining}: requests left in current window
  \item \texttt{X-RateLimit-Reset}: Unix timestamp of window reset
\end{itemize}
\textbf{429 response:} \texttt{\{"error": "Too many requests.", "retryAfter": N\}}
\end{secbox}

\subsection{IP Hashing}

Raw IP addresses are never stored. Before any database write:
$$
\text{ip\_hash} = \text{SHA-256}(\text{IP} \,\|\, \text{IP\_HASH\_SALT})
$$
where \texttt{IP\_HASH\_SALT} is a 32-byte random value from \texttt{openssl rand -hex 32},
stored only in the server-side environment. Hash is one-way and cannot be reversed.

\section{Mathematical Foundations: Five Core Theorems}
\label{sec:theorems}
% =============================================================================

\begin{theorem}[Banach Fixed-Point Contraction]
  Let $\mathcal{F}: \mathcal{H} \to \mathcal{H}$ on Hilbert space $\mathcal{H}$
  with $L_{\text{contract}} < 1$. Then $\exists!\; \theta^* \in \mathcal{H}$:
  $$
  \|\mathcal{F}^n(\theta_0) - \theta^*\| \;\leq\; L_{\text{contract}}^n \cdot \|\theta_0 - \theta^*\| \;\to\; 0
  $$
  \textit{Validated by: SIM-06 (L-contract sweep)}.
\end{theorem}

\begin{theorem}[Oja++ Manifold Convergence]
  Under $\eta_t^{\text{eff}} = \eta_0 / (1 + \hat{\sigma}^2_t / \delta_{\min})$
  on $\mathcal{V}_k(\mathbb{R}^d)$:
  $$
  \limsup_{t \to \infty} \mathbb{E}[V(\hat{W}_t)] \;\leq\; \frac{\eta_t B^4}{2\delta_k} + O(\sigma^2_{\text{noise}})
  $$
  \textit{Validated by: SIM-01, SIM-02}.
\end{theorem}

\begin{theorem}[Lyapunov GIM Stability]
  With $V(U_t) = \frac{1}{2}\|U_t - U^*\|^2$:
  $$
  \dot{V}(U_t) \;\leq\; -2\delta_k\,V(U_t)
  $$
  Exponential convergence to $U^*$ at rate $\geq 2\delta_k$.
  \textit{Validated by: SIM-05 (boundary damping)}.
\end{theorem}

\begin{theorem}[DASM Zero-Drift Rollback]
  Let $\theta_{\text{snapshot}}$ be the bitwise SRAM copy at each clock edge:
  $$
  \mathbb{E}\!\left[\|\theta_{\text{restored}} - \theta_t\|^2\right] = 0 \quad \text{vs} \quad
  \mathbb{E}[\|\theta_{\text{reversed}} - \theta_t\|^2] = 2\Theta\sigma^2 > 0
  $$
  \textit{Validated by: SIM-04, SIM-08}.
\end{theorem}

\begin{theorem}[SRC Discretisation Robustness]
  Open ball $B(x^*, r)$ with $r = \Delta - \rho/L_D$ is chattering-free:
  $$
  \forall\, x, y \in B(x^*, r): \quad s(x) = s(y)
  $$
  \textit{Validated by: SIM-03, SIM-12}.
\end{theorem}
